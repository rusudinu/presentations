\begin{frame}[eyebrow=Reference]{Layout widgets you need today}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \item \code{Column} and \code{Row}: arrange children vertically or horizontally.
        \item \code{Expanded}: takes remaining space inside a Row or Column.
        \item \code{ListView.builder}: lazy list of items; only builds what is visible.
        \item \code{Card} and \code{ListTile}: Material design containers.
        \item \code{Padding}: space around a child; \code{EdgeInsets} specifies how much.
      \end{itemize}
    \end{column}
    \begin{column}{0.41\textwidth}
      \begin{itemize}
        \item \code{SizedBox}: fixed width or height.
        \item \code{Container}: padding, margin, color, border in one widget.
        \item \code{Stack}: children drawn on top of each other.
        \item \code{Scaffold}: AppBar, body, FAB layout.
      \end{itemize}
      \begin{hint}
        Every one takes children, not pixels.
      \end{hint}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[eyebrow=Reference]{The DevTools tabs you need today}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \lead{Inspector}{The live widget tree. Select Widget Mode lets you tap a widget on screen. Track widget rebuilds counts how many times each widget rebuilt after an action.}
      \vskip 3mm
      \lead{Debugger}{Breakpoints, the call stack, and the Variables pane. Conditional breakpoints stop only when a condition is true.}
      \vskip 3mm
      \lead{Logging}{Everything \code{debugPrint} writes, plus framework events. Keep it open during debugging.}
    \end{column}
    \begin{column}{0.41\textwidth}
      \lead{Performance}{Frame timeline and overlay. Only meaningful in a \code{--profile} build.}
      \vskip 3mm
      \lead{Memory}{Heap snapshots. Find controllers you forgot to dispose.}
      \vskip 3mm
      \lead{Network}{Every HTTP request the app makes. You will need this in Lab 4.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile,eyebrow=Task III]{The three bugs to plant}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
\begin{dart}
// Bug 1: off-by-one
for (var i = 0; i <= _items.length; i++) {
  done += _items[i].done ? 1 : 0;
}

// Bug 2: no setState
void _toggle(int i) =>
    _items[i].done = !_items[i].done;

// Bug 3: null access
String? _filter;
final visible = _items.where(
    (t) => t.title.contains(_filter!));
\end{dart}
    \end{column}
    \begin{column}{0.41\textwidth}
      \begin{steps}
        \item Plant Bug 1, run until it throws.
        \item Set breakpoint, watch \code{i} and length.
        \item Fix. Plant Bug 2. Use conditional breakpoint.
        \item Plant Bug 3. Fix the null access.
      \end{steps}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[eyebrow=Reference]{What rebuilds, and why}
  \code{setState} marks one element dirty. Only that subtree rebuilds on the next frame. The framework reuses the element and State when the new widget has the same \code{runtimeType} and key.
  \vskip 3mm
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      Your list keeps scroll position when one item changes. Keys matter when items can be reordered or removed from the middle.
    \end{column}
    \begin{column}{0.41\textwidth}
      \kv{One frame}{16.7 ms at 60 Hz · 8.3 ms at 120 Hz}
      \vskip 2mm
      \begin{takeaway}{Rebuilt is not repainted.}
        Inspector counts rebuilds.
      \end{takeaway}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[eyebrow=Troubleshooting]{Layout errors and debugging issues}
  \vskip 2mm
  \trouble{RenderFlex overflowed by 42 pixels}{Wrap the growing child in Expanded.}
  \trouble{Vertical viewport was given unbounded height}{Wrap the ListView in Expanded.}
  \trouble{A dismissed Dismissible widget is still part of the tree}{Remove the item from the list in \code{onDismissed}.}
  \trouble{Breakpoints are hollow and never hit}{Start from the IDE (F5), not bare \code{flutter run}.}
  \trouble{The data changed, the screen did not}{You mutated state outside \code{setState}. No rebuilds.}
\end{frame}
